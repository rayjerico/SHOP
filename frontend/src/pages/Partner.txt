import { useEffect, useState } from "react";
import { ChevronLeft, ChevronRight } from "lucide-react";
import Jollibeat from "../assets/partner/jollibeat.png";
import Kisko from "../assets/partner/kisko.png";
import Malasme from "../assets/partner/malasme.png";
import Mangsinakal from "../assets/partner/mangsinakal.png";
import Nyek from "../assets/partner/nyek.png";
import Puregreen from "../assets/partner/puregreen.png";

const partners = [
  {
    name: "Jollibeat",
    website: "https://www.jollibee.com.ph/",
    image: Jollibeat,
  },
  {
    name: "Kisko",
    website: "https://www.cisco.com/",
    image: Kisko,
  },
  {
    name: "Malasme",
    website: "https://luckyme.ph/",
    image: Malasme,
  },
  {
    name: "Mangsinakal",
    website: "https://www.manginasal.ph/",
    image: Mangsinakal,
  },
  {
    name: "Ngek",
    website: "https://www.nike.com/ph/",
    image: Nyek,
  },
  {
    name: "Puregreen",
    website: "https://puregold.com.ph/",
    image: Puregreen,
  },
];

function Partner() {
  const [activeIndex, setActiveIndex] = useState(0);

  const move = (direction) => {
    setActiveIndex(
      (current) => (current + direction + partners.length) % partners.length,
    );
  };

  useEffect(() => {
    const timer = window.setInterval(() => {
      setActiveIndex((current) => (current + 1) % partners.length);
    }, 4500);
    return () => window.clearInterval(timer);
  }, []);

  return (
    <section className="overflow-hidden bg-[#10275a] py-14 text-white sm:py-16 lg:min-h-[405px] lg:py-[58px]">
      <div className="mx-auto max-w-[1322px] px-5 sm:px-10 lg:px-0">
        <div className="flex items-center justify-between">
          <h2 className="text-[32px] font-semibold tracking-tight sm:text-[40px]">
            Our Partner
          </h2>
          <div className="hidden gap-2 sm:flex">
            <button
              type="button"
              onClick={() => move(-1)}
              aria-label="Previous partners"
              className="rounded-full border border-white/40 p-2 transition hover:bg-white hover:text-[#10275a]"
            >
              <ChevronLeft size={20} />
            </button>
            <button
              type="button"
              onClick={() => move(1)}
              aria-label="Next partners"
              className="rounded-full border border-white/40 p-2 transition hover:bg-white hover:text-[#10275a]"
            >
              <ChevronRight size={20} />
            </button>
          </div>
        </div>

        <div
          className="mt-8 overflow-visible"
          aria-roledescription="carousel"
          aria-label="Partner websites"
        >
          <div
            className="flex gap-12 transition-transform duration-700 ease-out"
            style={{
              transform: `translateX(calc(-${activeIndex} * (298px + 48px)))`,
            }}
          >
            {partners.map((partner, index) => (
              <a
                key={`${partner.name}-${index}`}
                href={partner.website}
                target="_blank"
                rel="noreferrer"
                aria-label={`Visit ${partner.name}`}
                className="block w-[298px] shrink-0 overflow-hidden rounded-[10px] bg-white transition duration-300 hover:-translate-y-1 hover:shadow-xl focus:outline-none focus:ring-2 focus:ring-white focus:ring-offset-4 focus:ring-offset-[#10275a] sm:w-[298px]"
              >
                <img
                  src={partner.image}
                  alt={`${partner.name} logo`}
                  className="h-[153px] w-full object-contain p-3"
                />
              </a>
            ))}
          </div>
        </div>

        <div className="mt-7 flex items-center justify-center gap-2 sm:hidden">
          {partners.map((partner, index) => (
            <button
              key={partner.name}
              type="button"
              aria-label={`Show partner ${index + 1}`}
              aria-current={index === activeIndex}
              onClick={() => setActiveIndex(index)}
              className={`h-2 rounded-full transition-all ${index === activeIndex ? "w-6 bg-white" : "w-2 bg-white/40"}`}
            />
          ))}
        </div>
      </div>
    </section>
  );
}

export default Partner;

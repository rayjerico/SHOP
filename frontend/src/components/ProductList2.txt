import { useEffect, useState } from "react";
import axios from "axios";
import { Link } from "react-router-dom";
import { BASE_URL } from "../api/base";

const ProductList = () => {
  const [products, setProducts] = useState([]);
  const [showAll, setShowAll] = useState(false);

  useEffect(() => {
    let ignore = false;

    const fetchProducts = async () => {
      try {
        const response = await axios.get(`${BASE_URL}/api/products/`);
        const data = response?.data;

        const normalizedProducts = Array.isArray(data)
          ? data
          : Array.isArray(data?.results)
            ? data.results
            : [];

        if (!ignore) {
          setProducts(normalizedProducts);
        }
      } catch (err) {
        console.error("Failed to fetch products:", err);

        if (!ignore) {
          setProducts([]);
        }
      }
    };

    void fetchProducts();

    return () => {
      ignore = true;
    };
  }, []);

  const safeProducts = Array.isArray(products) ? products : [];
  const visibleProducts = showAll ? safeProducts : safeProducts.slice(0, 6);

  const getImageUrl = (image) => {
    if (!image) return "";

    if (typeof image === "string") {
      return image.startsWith("http") ? image : `${BASE_URL}${image}`;
    }

    if (typeof image === "object" && image !== null) {
      const url = typeof image.url === "string" ? image.url : "";
      return url.startsWith("http") ? url : `${BASE_URL}${url}`;
    }

    return "";
  };

  return (
    <section className="px-6 py-16">
      <h2 className="mb-12 text-center text-2xl font-extrabold tracking-wide text-[#10265A] md:text-3xl">
        PRODUCT LIST
      </h2>

      <div className="mx-auto grid max-w-6xl grid-cols-1 gap-8 sm:grid-cols-2 lg:grid-cols-3">
        {visibleProducts.map((item) => {
          const safeItem = item || {};
          const productName =
            safeItem.product_name || safeItem.name || "Product";
          const productPrice = safeItem.product_price ?? safeItem.price ?? 0;
          const brand = safeItem.brand || "Brand";
          const imageUrl = getImageUrl(safeItem.image);

          return (
            <Link
              to={`/products/${safeItem.id ?? ""}`}
              key={safeItem.id ?? productName}
            >
              <article className="flex h-full flex-col overflow-hidden rounded-lg border border-gray-200 bg-white shadow-sm transition hover:shadow-lg">
                <img
                  src={imageUrl}
                  alt={productName}
                  className="h-70 w-full object-contain p-6"
                />

                <div className="flex flex-1 flex-col justify-between px-6 pb-6">
                  <div className="mb-1 flex items-start justify-between gap-4">
                    <p className="text-sm font-semibold text-gray-800">
                      {productName}
                    </p>
                    <p className="text-sm font-semibold text-gray-700">
                      ${Number(productPrice).toFixed(2)}
                    </p>
                  </div>

                  <p className="mb-4 text-xs font-medium uppercase tracking-wide text-gray-500">
                    {brand}
                  </p>

                  <button
                    type="button"
                    className="self-end rounded bg-[#10265A] px-4 py-1.5 text-xs font-semibold tracking-wide text-white transition hover:bg-[#0b1d45]"
                  >
                    BUY
                  </button>
                </div>
              </article>
            </Link>
          );
        })}
      </div>

      {safeProducts.length > 6 && !showAll && (
        <div className="mt-14 flex justify-center">
          <button
            type="button"
            onClick={() => setShowAll(true)}
            className="rounded bg-[#10265A] px-8 py-2.5 text-sm font-semibold tracking-wide text-white transition hover:bg-[#0b1d45]"
          >
            View all products
          </button>
        </div>
      )}
    </section>
  );
};

export default ProductList;
